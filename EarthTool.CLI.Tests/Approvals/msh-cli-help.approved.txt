# earthtool --help
USAGE:
    earthtool [OPTIONS] <COMMAND>

OPTIONS:
    -h, --help    Prints help information

COMMANDS:
    wd                     WD archive management commands
    msh                    MSH and glTF interchange commands
    tex <InputFilePath>
    par                    PAR parameter file commands

# earthtool msh --help
DESCRIPTION:
MSH and glTF interchange commands

USAGE:
    earthtool msh [OPTIONS] <COMMAND>

OPTIONS:
    -h, --help    Prints help information

COMMANDS:
    export <INPUT>    Export supported MSH assets to GLB or separate glTF
    import <INPUT>    Create MSH assets from GLB or separate glTF

# earthtool msh import --help
DESCRIPTION:
Create MSH assets from GLB or separate glTF

USAGE:
    earthtool msh import <INPUT> [OPTIONS]

ARGUMENTS:
    <INPUT>    One or more concrete input files or file patterns

OPTIONS:
    -h, --help                  Prints help information
    -o, --output <DIRECTORY>    Destination directory. Defaults to the input
                                directory
        --report <FILE>         Optional versioned machine report path
        --plan <FILE>           Optional version-2 typed import plan
